```python
def parse_sjson(s: str):
    """
    Parse a JSON string (objects, arrays, strings with escapes, numbers,
    true, false, null) and return the corresponding Python value.
    No use of json module or eval.
    """
    i = 0
    n = len(s)

    def skip_ws():
        nonlocal i
        while i < n and s[i] in ' \t\n\r':
            i += 1

    def parse_value():
        skip_ws()
        if i >= n:
            raise ValueError('Unexpected end of input')
        ch = s[i]
        if ch == '"':
            return parse_string()
        if ch == '{':
            return parse_object()
        if ch == '[':
            return parse_array()
        if s.startswith('true', i):
            i_advance(4)
            return True
        if s.startswith('false', i):
            i_advance(5)
            return False
        if s.startswith('null', i):
            i_advance(4)
            return None
        return parse_number()

    def i_advance(k):
        nonlocal i
        i += k

    def parse_string():
        nonlocal i
        i_advance(1)                     # skip opening quote
        out = []
        while True:
            if i >= n:
                raise ValueError('Unterminated string')
            ch = s[i]
            if ch == '"':
                i_advance(1)
                break
            if ch == '\\':
                i_advance(1)
                if i >= n:
                    raise ValueError('Unterminated escape')
                esc = s[i]
                i_advance(1)
                if esc == '"':
                    out.append('"')
                elif esc == '\\':
                    out.append('\\')
                elif esc == '/':
                    out.append('/')
                elif esc == 'b':
                    out.append('\b')
                elif esc == 'f':
                    out.append('\f')
                elif esc == 'n':
                    out.append('\n')
                elif esc == 'r':
                    out.append('\r')
                elif esc == 't':
                    out.append('\t')
                elif esc == 'u':
                    if i + 4 > n:
                        raise ValueError('Invalid \\u escape')
                    hex_part = s[i:i+4]
                    if not all(c in '0123456789abcdefABCDEF' for c in hex_part):
                        raise ValueError('Invalid \\u escape')
                    out.append(chr(int(hex_part, 16)))
                    i_advance(4)
                else:
                    raise ValueError(f'Invalid escape character \\{esc}')
            else:
                out.append(ch)
                i_advance(1)
        return ''.join(out)

    def parse_number():
        nonlocal i
        start = i
        if s[i] == '-':
            i_advance(1)
        if i < n and s[i] == '0':
            i_advance(1)
        else:
            if i >= n or not s[i].isdigit():
                raise ValueError('Invalid number')
            while i < n and s[i].isdigit():
                i_advance(1)
        if i < n and s[i] == '.':
            i_advance(1)
            if i >= n or not s[i].isdigit():
                raise ValueError('Invalid fractional part')
            while i < n and s[i].isdigit():
                i_advance(1)
        if i < n and s[i] in 'eE':
            i_advance(1)
            if i < n and s[i] in '+-':
                i_advance(1)
            if i >= n or not s[i].isdigit():
                raise ValueError('Invalid exponent')
            while i < n and s[i].isdigit():
                i_advance(1)
        num_str = s[start:i]
        return int(num_str) if ('.' not in num_str and 'e' not in num_str and 'E' not in num_str) else float(num_str)

    def parse_array():
        nonlocal i
        i_advance(1)                     # skip '['
        skip_ws()
        arr = []
        if i < n and s[i] == ']':
            i_advance(1)
            return arr
        while True:
            arr.append(parse_value())
            skip_ws()
            if i >= n:
                raise ValueError('Unterminated array')
            if s[i] == ',':
                i_advance(1)
                skip_ws()
                continue
            if s[i] == ']':
                i_advance(1)
                break
            raise ValueError('Expected , or ] in array')
        return arr

    def parse_object():
        nonlocal i
        i_advance(1)                     # skip '{'
        skip_ws()
        obj = {}
        if i < n and s[i] == '}':
            i_advance(1)
            return obj
        while True:
            skip_ws()
            if i >= n or s[i] != '"':
                raise ValueError('Object keys must be strings')
            key = parse_string()
            skip_ws()
            if i >= n or s[i] != ':':
                raise ValueError('Missing colon after key')
            i_advance(1)
            value = parse_value()
            obj[key] = value
            skip_ws()
            if i >= n:
                raise ValueError('Unterminated object')
            if s[i] == ',':
                i_advance(1)
                continue
            if s[i] == '}':
                i_advance(1)
                break
            raise ValueError('Expected , or } in object')
        return obj

    result = parse_value()
    skip_ws()
    if i != n:
        raise ValueError('Extra data after valid JSON')
    return result
```
