def base_to_int(s, base):
    if not isinstance(base, int) or base < 2 or base > 36:
        raise ValueError("base must be an integer between 2 and 36")
    if not s:
        raise ValueError("input string cannot be empty")
    s = s.strip()
    if not s:
        raise ValueError("input string cannot be empty")
    digits = "0123456789abcdefghijklmnopqrstuvwxyz"
    value = 0
    for ch in s.lower():
        if ch not in digits:
            raise ValueError(f"invalid character '{ch}'")
        digit_val = digits.index(ch)
        if digit_val >= base:
            raise ValueError(f"digit '{ch}' not valid for base {base}")
        value = value * base + digit_val
    return value
