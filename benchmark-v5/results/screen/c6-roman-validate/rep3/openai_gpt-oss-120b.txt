def roman_to_int(s: str) -> int:
    """
    Convert a Roman numeral string to an integer.

    Args:
        s: Upper‑case Roman numeral (e.g., "MCMXLIV").

    Returns:
        Integer value of the numeral.

    Raises:
        ValueError: If the string is empty, contains illegal characters,
                    uses illegal repetitions, or has a non‑canonical form.
    """
    if not s:
        raise ValueError("Empty string is not a valid Roman numeral")

    # Valid symbols and their values
    values = {'I': 1, 'V': 5, 'X': 10, 'L': 50,
              'C': 100, 'D': 500, 'M': 1000}

    # Allowed subtractive pairs and the value they represent
    subtractives = {
        'IV': 4, 'IX': 9,
        'XL': 40, 'XC': 90,
        'CD': 400, 'CM': 900
    }

    # Check for illegal characters early
    for ch in s:
        if ch not in values:
            raise ValueError(f"Invalid character '{ch}' in Roman numeral")

    # Helper to check repetitions (no symbol > 3 times in a row,
    # and V, L, D may never repeat)
    def check_repetitions(num_str: str) -> None:
        repeat_count = 1
        last_char = num_str[0]
        for ch in num_str[1:]:
            if ch == last_char:
                repeat_count += 1
                if repeat_count > 3:
                    raise ValueError("Symbol repeated more than three times")
                if ch in ('V', 'L', 'D'):
                    raise ValueError(f"Symbol '{ch}' may not be repeated")
            else:
                repeat_count = 1
                last_char = ch

    # First, ensure the string respects repetition rules
    check_repetitions(s)

    i = 0
    total = 0
    while i < len(s):
        # Look ahead for a subtractive pair
        if i + 1 < len(s):
            pair = s[i] + s[i + 1]
            if pair in subtractives:
                total += subtractives[pair]
                i += 2
                continue

        # Otherwise, single symbol
        total += values[s[i]]
        i += 1

    # After conversion, verify that the numeral is in canonical form
    # by re‑encoding the integer and comparing with the original.
    # This catches cases like "IL", "IIX", "VX", etc.
    def int_to_roman(num: int) -> str:
        mapping = [
            (1000, "M"), (900, "CM"), (500, "D"), (400, "CD"),
            (100, "C"), (90, "XC"), (50, "L"), (40, "XL"),
            (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")
        ]
        res = []
        for val, sym in mapping:
            while num >= val:
                res.append(sym)
                num -= val
        return "".join(res)

    canonical = int_to_roman(total)
    if canonical != s:
        raise ValueError("Non‑canonical Roman numeral representation")

    return total
