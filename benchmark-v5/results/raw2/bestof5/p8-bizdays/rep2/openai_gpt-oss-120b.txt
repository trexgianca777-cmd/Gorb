def add_business_days(date_str: str, n: int) -> str:
    # ----- helpers: convert civil date <-> days since 1970-01-01 -----
    def days_from_civil(y: int, m: int, d: int) -> int:
        # Howard Hinnant's algorithm, works for all Gregorian dates
        y -= m <= 2
        era = (y >= 0 and y) // 400 - (y < 0) // 400
        yoe = y - era * 400                     # [0, 399]
        doy = (153 * (m + (9 if m < 3 else -3)) + 2) // 5 + d - 1  # [0, 365]
        doe = yoe * 365 + yoe // 4 - yoe // 100 + doy               # [0, 146096]
        return era * 146097 + doe - 719468  # days since 1970‑01‑01

    def civil_from_days(z: int):
        z += 719468
        era = (z >= 0 and z) // 146097 - (z < 0) // 146097
        doe = z - era * 146097                     # [0, 146096]
        yoe = (doe - doe // 1460 + doe // 36524 - doe // 146096) // 365  # [0,399]
        y = yoe + era * 400
        doy = doe - (365 * yoe + yoe // 4 - yoe // 100)                # [0,365]
        mp = (5 * doy + 2) // 153                                      # [0,11]
        d = doy - (153 * mp + 2) // 5 + 1
        m = mp + 3 if mp < 10 else mp - 9
        y += m <= 2
        return y, m, d

    # ----- parse input -----
    y, m, d = map(int, date_str.split('-'))
    days = days_from_civil(y, m, d)

    # weekday: Monday=0 … Sunday=6 (1970‑01‑01 was Thursday)
    weekday = (days + 3) % 7

    # ----- add business days -----
    if n == 0:
        return date_str

    days_to_add = 0
    if n > 0:
        weeks, extra = divmod(n, 5)
        days_to_add = weeks * 7
        while extra:
            days_to_add += 1
            if (weekday + days_to_add) % 7 < 5:   # Mon‑Fri
                extra -= 1
    else:                     # n < 0
        # Python's divmod gives negative remainder, we need separate handling
        weeks = n // 5                     # floor division (negative)
        extra = n % 5                      # negative or zero
        days_to_add = weeks * 7
        while extra:
            days_to_add -= 1
            if (weekday + days_to_add) % 7 < 5:
                extra += 1

    new_days = days + days_to_add
    ny, nm, nd = civil_from_days(new_days)
    return f"{ny:04d}-{nm:02d}-{nd:02d}"
