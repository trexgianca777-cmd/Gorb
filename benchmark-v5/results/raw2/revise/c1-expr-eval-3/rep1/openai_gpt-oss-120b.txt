```python
import re

def evaluate(s: str) -> float:
    tokens = re.findall(r'\s*(\d+|\*\*|[+\-*/()])', s)
    pos = 0

    def peek():
        return tokens[pos] if pos < len(tokens) else None

    def get():
        nonlocal pos
        tok = tokens[pos]
        pos += 1
        return tok

    # ---- parsing ----
    def parse_expression():
        node = parse_term()
        while peek() in ('+', '-'):
            op = get()
            node = (op, node, parse_term())
        return node

    def parse_term():
        node = parse_factor()
        while peek() in ('*', '/'):
            op = get()
            node = (op, node, parse_factor())
        return node

    def parse_factor():
        # factor handles unary minus
        if peek() == '-':
            get()
            return ('neg', parse_factor())
        return parse_power()

    def parse_power():
        left = parse_atom()
        if peek() == '**':
            get()
            # right side of ** may have unary operators (factor)
            right = parse_factor()
            return ('**', left, right)
        return left

    def parse_atom():
        if peek() == '(':
            get()
            node = parse_expression()
            if peek() != ')':
                raise SyntaxError("Missing ')'")
            get()
            return node
        tok = get()
        if re.fullmatch(r'\d+', tok):
            return ('num', float(tok))
        raise SyntaxError(f'Unexpected token: {tok}')

    # ---- evaluation ----
    def eval_node(node):
        typ = node[0]
        if typ == 'num':
            return node[1]
        if typ == 'neg':
            return -eval_node(node[1])
        op, left, right = node
        a, b = eval_node(left), eval_node(right)
        if op == '+':
            return a + b
        if op == '-':
            return a - b
        if op == '*':
            return a * b
        if op == '/':
            return a / b
        if op == '**':
            return a ** b
        raise ValueError(f'Unknown operator {op}')

    ast = parse_expression()
    if pos != len(tokens):
        raise SyntaxError('Extra input')
    return eval_node(ast)
```
