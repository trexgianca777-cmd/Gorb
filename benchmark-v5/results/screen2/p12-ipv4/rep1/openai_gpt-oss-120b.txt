def ipv4_to_int(ip: str) -> int:
    parts = ip.split('.')
    if len(parts) != 4:
        raise ValueError("IPv4 address must have exactly 4 parts")
    result = 0
    for part in parts:
        if not part:
            raise ValueError("Empty octet in IPv4 address")
        if not part.isdigit():
            raise ValueError(f"Non-numeric octet: {part!r}")
        if len(part) > 1 and part[0] == '0':
            raise ValueError(f"Octet with leading zero: {part!r}")
        octet = int(part)
        if octet > 255:
            raise ValueError(f"Octet out of range (0-255): {octet}")
        result = (result << 8) | octet
    return result
